-- =========================================================
-- Barangay Health System (Improved Relational Schema)
-- - Role-based users (patient/health_worker/admin)
-- - File uploads (profile + identity) stored as metadata rows
-- - Security questions stored separately with hashed answers
-- - Patient identity verification support
-- =========================================================

CREATE DATABASE IF NOT EXISTS barangay_health_system CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE barangay_health_system;

-- Optional: for rebuilds during development
-- DROP TABLE IF EXISTS patient_medications, patient_diseases, medicine_inventory, medications, diseases, patient_verifications, user_security_questions, files, patients, health_workers, admins, users;

CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    email VARCHAR(120) NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('patient','health_worker','admin') NOT NULL,
    created_by_user_id INT NULL, -- admin who created the account (health_worker)
    is_active TINYINT(1) NOT NULL DEFAULT 1,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_users_username (username),
    UNIQUE KEY uq_users_email (email),
    KEY idx_users_role (role),
    KEY idx_users_created_by (created_by_user_id),
    CONSTRAINT fk_users_created_by FOREIGN KEY (created_by_user_id) REFERENCES users(user_id)
        ON DELETE SET NULL ON UPDATE CASCADE
);

-- Patient personal details
CREATE TABLE patients (
    patient_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,

    surname VARCHAR(50) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    middlename VARCHAR(50) NULL,

    date_of_birth DATE NOT NULL,
    address TEXT NOT NULL,
    contact_number VARCHAR(20) NOT NULL,

    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_patients_user_id (user_id),
    KEY idx_patients_dob (date_of_birth),
    CONSTRAINT fk_patients_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- Health worker personal details
CREATE TABLE health_workers (
    worker_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,

    surname VARCHAR(50) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    middlename VARCHAR(50) NULL,

    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_health_workers_user_id (user_id),
    CONSTRAINT fk_health_workers_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- Admin personal details (no public registration)
CREATE TABLE admins (
    admin_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,

    surname VARCHAR(50) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    middlename VARCHAR(50) NULL,

    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_admins_user_id (user_id),
    CONSTRAINT fk_admins_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- Files table: store upload metadata (actual bytes stored in filesystem/object storage)
CREATE TABLE files (
    file_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    purpose ENUM('profile_image','patient_identity') NOT NULL,
    original_name VARCHAR(255) NOT NULL,
    storage_name VARCHAR(255) NOT NULL,
    storage_path VARCHAR(500) NOT NULL,
    mime_type VARCHAR(100) NOT NULL,
    size_bytes BIGINT NOT NULL,
    uploaded_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    KEY idx_files_user_purpose (user_id, purpose),
    CONSTRAINT fk_files_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- Security questions: answers stored as bcrypt hashes
-- - Patients: optional (0..3)
-- - Health workers: mandatory at least 1
-- - Admins: mandatory 3 (second step after password)
CREATE TABLE user_security_questions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    question_no TINYINT NOT NULL, -- 1..3
    question_text VARCHAR(255) NOT NULL,
    answer_hash VARCHAR(255) NOT NULL,
    is_required TINYINT(1) NOT NULL DEFAULT 0,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uq_user_question_no (user_id, question_no),
    CONSTRAINT fk_user_security_questions_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- Patient identity verification workflow (upload a real image file)
CREATE TABLE patient_verifications (
    verification_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    status ENUM('pending','approved','rejected') NOT NULL DEFAULT 'pending',
    submitted_file_id INT NULL, -- files.file_id with purpose='patient_identity'
    notes VARCHAR(255) NULL,
    submitted_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reviewed_by_user_id INT NULL,
    reviewed_at TIMESTAMP NULL,

    UNIQUE KEY uq_patient_verification_user (user_id),
    CONSTRAINT fk_patient_verifications_user FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_patient_verifications_file FOREIGN KEY (submitted_file_id) REFERENCES files(file_id)
        ON DELETE SET NULL ON UPDATE CASCADE,
    CONSTRAINT fk_patient_verifications_reviewer FOREIGN KEY (reviewed_by_user_id) REFERENCES users(user_id)
        ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE diseases (
    disease_id INT AUTO_INCREMENT PRIMARY KEY,
    disease_name VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE patient_diseases (
    id INT AUTO_INCREMENT PRIMARY KEY,

    patient_id INT,
    disease_id INT,
    diagnosed_date DATE,

    FOREIGN KEY (patient_id) REFERENCES patients(patient_id),
    FOREIGN KEY (disease_id) REFERENCES diseases(disease_id)
);

CREATE TABLE medications (
    medication_id INT AUTO_INCREMENT PRIMARY KEY,
    medication_name VARCHAR(100) UNIQUE
);

CREATE TABLE patient_medications (
    id INT AUTO_INCREMENT PRIMARY KEY,

    patient_id INT,
    medication_id INT,

    dosage VARCHAR(50),
    start_date DATE,
    end_date DATE,

    FOREIGN KEY (patient_id) REFERENCES patients(patient_id),
    FOREIGN KEY (medication_id) REFERENCES medications(medication_id)
);

CREATE TABLE medicine_inventory (
    inventory_id INT AUTO_INCREMENT PRIMARY KEY,

    medication_id INT,
    quantity INT DEFAULT 0,

    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (medication_id) REFERENCES medications(medication_id)
);
